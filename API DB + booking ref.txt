--
-- Table structure for table `reservation`
-- One row per seat; rows booked together share the same booking_ref
--

DROP TABLE IF EXISTS `reservation`;

CREATE TABLE `reservation` (
  `id` int NOT NULL AUTO_INCREMENT,
  `booking_ref` CHAR(36) NOT NULL,
  `showtime_id` int NOT NULL,
  `user_id` int NOT NULL,
  `seat_label` VARCHAR(5) NOT NULL,
  `price` DECIMAL(8,2) NOT NULL DEFAULT 0.00,
  PRIMARY KEY (`id`),
  KEY `ix_reservation_booking_ref` (`booking_ref`),
  CONSTRAINT `unique_seat_showtime` UNIQUE (`showtime_id`, `seat_label`),
  CONSTRAINT `fk_user_reservation` FOREIGN KEY (`user_id`) REFERENCES `user`(`id`) ON DELETE NO ACTION ON UPDATE NO ACTION,
  CONSTRAINT `fk_showtimes_reservation` FOREIGN KEY (`showtime_id`) REFERENCES `showtime`(`id`)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4;

DROP TABLE IF EXISTS `refresh_token`;

CREATE TABLE refresh_token (
  id INT NOT NULL AUTO_INCREMENT,
  token_hash CHAR(64) NOT NULL,
  username VARCHAR(50) NOT NULL,
  family_id CHAR(36) NOT NULL,
  expires_at DATETIME(6) NOT NULL,
  revoked TINYINT(1) NOT NULL DEFAULT 0,
  PRIMARY KEY (id),
  UNIQUE KEY uq_rt_hash (token_hash),
  KEY ix_rt_family (family_id),
  KEY ix_rt_username (username),
  CONSTRAINT fk_rt_userr FOREIGN KEY (username)
    REFERENCES user(username) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
