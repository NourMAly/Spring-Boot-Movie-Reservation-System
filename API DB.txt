-- Table structure for table `showtime`

DROP TABLE IF EXISTS `showtime`;

CREATE TABLE `showtime` (

  `hall_number` int DEFAULT NULL,
  `start_time` time DEFAULT NULL,
  `movie_id` int DEFAULT NULL,
  `show_date` DATE NOT NULL DEFAULT(CURRENT_DATE),
  `price` DECIMAL(8,2) NOT NULL DEFAULT 0.00,
  PRIMARY KEY (`id`),
  CONSTRAINT `fk_movie_showtime` FOREIGN KEY (`movie_id`) REFERENCES `movie` (`id`) ON DELETE NO ACTION ON UPDATE NO ACTION,
  CONSTRAINT `fk_halls_showtimes` FOREIGN KEY (`hall_number`) references `hall`(`id`)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4;

-- Table structure for table `reservation`

DROP TABLE IF EXISTS `reservation`;

CREATE TABLE `reservation` (
  `id` int NOT NULL AUTO_INCREMENT,
  `showtime_id` int DEFAULT NULL,
  `customer_id` int DEFAULT NULL,

  PRIMARY KEY (`id`),
  CONSTRAINT `fk_customer_reservation` FOREIGN KEY (`customer_id`) REFERENCES `customer`(`id`) ON DELETE NO ACTION ON UPDATE NO ACTION,

  CONSTRAINT `fk_showtimes_reservation` FOREIGN KEY (`showtime_id`) references `showtime`(`id`)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4;
